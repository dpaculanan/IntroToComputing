const conversionOptions = {
  'c-to-f': {
    from: '°C', to: '°F', formula: 'Formula: °F = (°C × 9/5) + 32',
    convert: (value) => value * 9 / 5 + 32
  },
  'f-to-c': {
    from: '°F', to: '°C', formula: 'Formula: °C = (°F − 32) × 5/9',
    convert: (value) => (value - 32) * 5 / 9
  },
  'm-to-ft': {
    from: 'm', to: 'ft', formula: 'Formula: feet = meters × 3.28084',
    convert: (value) => value * 3.28084
  },
  'ft-to-m': {
    from: 'ft', to: 'm', formula: 'Formula: meters = feet × 0.3048',
    convert: (value) => value * 0.3048
  }
};

const taxableIncome = Number(incomeInput.value);
let basicTax = 0;
let rate = 0;
let bracketFloor = 0;
let formula = '0%';

if (taxableIncome <= 250000) {
  formula = '0%';
} else if (taxableIncome <= 400000) {
  bracketFloor = 250000;
  rate = 0.20;
  formula = '20% of excess over ₱250,000';
} else if (taxableIncome <= 800000) {
  basicTax = 30000;
  bracketFloor = 400000;
  rate = 0.25;
  formula = '₱30,000 + 25% of excess over ₱400,000';

  basicTax = 130000;
  bracketFloor = 800000;
  rate = 0.30;

  basicTax = 490000;

  basicTax = 2410000;

Three ways to count through the same numbers.

const count = Number(input.value);
let factorial = 1;
let factor = 1;
while (factor <= count) {
  factorial *= factor;
  factor += 1;
}

let sum = 0;
let current = 1;
do {
  sum += current;
  current += 1;
} while (current <= count);

let total = 0;
for (let number = 1; number <= count; number += 1) {
  total += number;
}
const average = total / count;
